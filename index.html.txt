  <script>
    const formView = document.getElementById("formView");
    const receiptView = document.getElementById("receiptView");
    const sendForm = document.getElementById("sendForm");
    const errorBox = document.getElementById("errorBox");

    const displayAmount = document.getElementById("displayAmount");
    const displayAddress = document.getElementById("displayAddress");
    const displayNetworkText = document.getElementById("displayNetworkText");
    const displayDate = document.getElementById("displayDate");

    function getFormattedDate() {
      const now = new Date();
      const pad = (n) => String(n).padStart(2, "0");
      const day = pad(now.getDate());
      const month = pad(now.getMonth() + 1);
      const year = now.getFullYear();
      const hours = pad(now.getHours());
      const minutes = pad(now.getMinutes());
      const seconds = pad(now.getSeconds());
      return `Date: ${day}/${month}/${year}, ${hours}:${minutes}:${seconds}`;
    }

    function formatDisplayAmount(value, currency) {
      const num = Number(value);
      if (currency === "USD") {
        return "$" + num.toLocaleString("en-US", {
          minimumFractionDigits: 2,
          maximumFractionDigits: 2
        });
      }
      return num.toLocaleString("en-US", {
        minimumFractionDigits: 2,
        maximumFractionDigits: 8
      }) + " BTC";
    }

    sendForm.addEventListener("submit", (event) => {
      event.preventDefault();
      errorBox.textContent = "";

      const address = document.getElementById("addressInput").value.trim();
      const currency = document.getElementById("currencyInput").value;
      const amount = document.getElementById("amountInput").value.trim();

      if (!address || Number(amount) <= 0 || Number.isNaN(Number(amount))) {
        errorBox.textContent = "Enter a valid wallet address and amount.";
        return;
      }

      displayAmount.textContent = formatDisplayAmount(amount, currency);
      displayAddress.textContent = address;
      displayNetworkText.textContent = "BTC on";
      displayDate.textContent = getFormattedDate();

      formView.classList.add("hidden");
      receiptView.classList.remove("hidden");
    });

    // Double-click or tap on the confirmation screen to go back to the form
    receiptView.addEventListener("dblclick", () => {
      receiptView.classList.add("hidden");
      formView.classList.remove("hidden");
    });
  </script>
